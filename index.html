/** * ═══════════════════════════════════════════════════════════════ * MUHS 50 Years — Registration Backend (Google Apps Script) * Sheets = database, Drive = photo storage * ═══════════════════════════════════════════════════════════════ * * SETUP (একবারই করতে হবে): * * 1) নতুন একটা Google Sheet বানান (sheets.new) — নাম দিন যা খুশি, * যেমন "MUHS 50 Years Registrations" * * 2) সেই Sheet-এর URL থেকে SPREADSHEET_ID কপি করুন: * https://docs.google.com/spreadsheets/d/ ⬅️এই_অংশটুকু⬅️ /edit * docs.google.com/spreadsheets/d/1AbCxyz.......XyZ/edit * └────── এই ID টুকু ──────┘ * নিচে SPREADSHEET_ID = '...' এর মধ্যে সেটা বসান। * * 3) ওই Sheet-এর ভেতর থেকেই: Extensions → Apps Script → এই পুরো * কোড পেস্ট করুন (এভাবে করলে Sheet-এর সাথে script bound থাকবে, * তবে SPREADSHEET_ID দেওয়া থাকায় standalone script দিয়েও কাজ করবে) * * 4) Project Settings (⚙️) → Script Properties → নতুন property: * key: ADMIN_KEY * value: (আপনার নিজের গোপন পাসওয়ার্ড, যেমন: MUHS2026_Secret_92x) * * 5) Deploy → New deployment → type: Web app * Execute as: Me * Who has access: Anyone * "Deploy" চাপার পর প্রথমবার permission চাইবে — Authorize করুন, * "Advanced" → "Go to project (unsafe)" চেপে নিজের Google * account দিয়ে অনুমতি দিন (এটা আপনার নিজের script, নিরাপদ)। * * 6) যে URL পাবেন সেটা index.html এর GAS_URL এ বসান * * 7) টেস্ট: ব্রাউজারে URL + "?action=getAll" খুলুন, যেমন: * https://script.google.com/macros/s/AKfycb.../exec?action=getAll * দেখুন {"success":true,"total":0,...} আসে কিনা। * যদি error/HTML page আসে, সেটার লেখাটা কপি করে রাখুন — debug-এ লাগবে। * * ⚠️ কোড পরিবর্তন করার পর প্রতিবার নতুন করে Deploy করতে হবে * (Deploy → Manage deployments → ✏️ Edit → Version: New version → Deploy) * শুধু কোড সেভ করলেই পুরনো deployment আপডেট হয় না! */ const SHEET_NAME = 'Registrations'; const PHOTO_FOLDER_NAME = 'MUHS_50years_Photos'; // ⚠️ এটা অবশ্যই বসাতে হবে — উপরে দেখুন "SPREADSHEET_ID কোথায় পাবেন" const SPREADSHEET_ID = 'PASTE_YOUR_SPREADSHEET_ID_HERE'; // ── Sheet columns ── // A:Timestamp B:ID C:UniqueID D:Name E:Phone F:Email G:Type H:Gender // I:Batch J:District K:Guests L:Message M:Date N:PhotoURL function getSheet_() { if (!SPREADSHEET_ID || SPREADSHEET_ID === 'PASTE_YOUR_SPREADSHEET_ID_HERE') { throw new Error('SPREADSHEET_ID সেট করা হয়নি! Code.gs এর উপরের দিকে SPREADSHEET_ID বসান।'); } const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['Timestamp','ID','UniqueID','Name','Phone','Email','Type','Gender','Batch','District','Guests','Message','Date','PhotoURL']); sheet.setFrozenRows(1); } return sheet; } function getPhotoFolder_() { const folders = DriveApp.getFoldersByName(PHOTO_FOLDER_NAME); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(PHOTO_FOLDER_NAME); } function isValidAdminKey_(key) { const adminKey = PropertiesService.getScriptProperties().getProperty('ADMIN_KEY'); return !!adminKey && !!key && key === adminKey; } // ═══════════ GET ═══════════ function doGet(e) { try { const action = e.parameter.action; const key = e.parameter.key || ''; if (action === 'getAll') { return jsonOut_(getAllRegistrations_(isValidAdminKey_(key))); } return jsonOut_({ success: false, error: 'unknown_action' }); } catch (err) { return jsonOut_({ success: false, error: 'server_error', detail: String(err) }); } } // ═══════════ POST ═══════════ function doPost(e) { try { let payload; try { payload = JSON.parse(e.postData.contents); } catch (err) { return jsonOut_({ success: false, error: 'invalid_json' }); } if (payload.action === 'register') { return jsonOut_(registerUser_(payload)); } if (payload.imageData) { return jsonOut_(uploadPhoto_(payload)); } return jsonOut_({ success: false, error: 'unknown_request' }); } catch (err) { return jsonOut_({ success: false, error: 'server_error', detail: String(err) }); } } // ═══════════ REGISTER ═══════════ function registerUser_(data) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = getSheet_(); const values = sheet.getDataRange().getValues(); // Server-side duplicate phone check (client check can be bypassed) for (let i = 1; i < values.length; i++) { if (String(values[i][4]).trim() === String(data.phone).trim()) { return { success: false, error: 'duplicate_phone' }; } } sheet.appendRow([ new Date(), data.id || '', data.uniqueId || '', data.name || '', data.phone || '', data.email || '', data.type || '', data.gender || '', data.batch || '', data.district || '', data.guests || '', data.message || '', data.date || '', '' ]); return { success: true, uniqueId: data.uniqueId }; } catch (err) { return { success: false, error: 'server_error', detail: String(err) }; } finally { lock.releaseLock(); } } // ═══════════ PHOTO UPLOAD ═══════════ function uploadPhoto_(data) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const matches = String(data.imageData).match(/^data:(image\/\w+);base64,(.+)$/); if (!matches) return { success: false, error: 'bad_image_format' }; const mime = matches[1]; const bytes = Utilities.base64Decode(matches[2]); const blob = Utilities.newBlob(bytes, mime, (data.uniqueId || 'photo') + '.jpg'); const folder = getPhotoFolder_(); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); const url = 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w400'; // matching row-এ PhotoURL বসাও const sheet = getSheet_(); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][2]).trim() === String(data.uniqueId).trim()) { sheet.getRange(i + 1, 14).setValue(url); // column N break; } } return { success: true, url: url }; } catch (err) { return { success: false, error: 'server_error', detail: String(err) }; } finally { lock.releaseLock(); } } // ═══════════ GET ALL (public = summary only, admin = full list) ═══════════ function getAllRegistrations_(isAdmin) { const sheet = getSheet_(); const values = sheet.getDataRange().getValues(); const rows = values.slice(1).filter(r => r[2]); // skip blank rows const summary = { total: rows.length, alumni: 0, student: 0, teacher: 0, guest: 0, other: 0 }; const registrations = []; rows.forEach(row => { const type = row[6] || 'other'; if (summary[type] !== undefined) summary[type]++; else summary.other++; if (isAdmin) { registrations.push({ id: row[1], uniqueId: row[2], name: row[3], phone: row[4], email: row[5], type: row[6], gender: row[7], batch: row[8], district: row[9], guests: row[10], message: row[11], date: row[12], photo: row[13] || '' }); } }); return { success: true, total: rows.length, summary: summary, registrations: registrations }; } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
